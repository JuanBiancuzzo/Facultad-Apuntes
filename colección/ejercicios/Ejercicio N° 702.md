	~i~ & \theta & d & a & ~\alpha~ \\\hline
	1 & q_1 & 0 & a_1 & 0 \\\hline
	2 & q_2 & 0 & a_2 & 0 \\\hline
	3 & 0 & q_3 & 0 & 0 \\\hline
	4 & q_4 & 0 & 0 & 0 
\end{array} $$
